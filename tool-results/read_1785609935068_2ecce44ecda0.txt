   708→        function drawStar(cx, cy, spikes, outerR, innerR) {
   709→            let rot = Math.PI / 2 * 3;
   710→            const step = Math.PI / spikes;
   711→            ctx.beginPath();
   712→            ctx.moveTo(cx, cy - outerR);
   713→            for (let i = 0; i < spikes; i++) {
   714→                ctx.lineTo(cx + Math.cos(rot) * outerR, cy + Math.sin(rot) * outerR);
   715→                rot += step;
   716→                ctx.lineTo(cx + Math.cos(rot) * innerR, cy + Math.sin(rot) * innerR);
   717→                rot += step;
   718→            }
   719→            ctx.closePath();
   720→            ctx.fill();
   721→        }
   722→
   723→        function animateParticles() {
   724→            ctx.clearRect(0, 0, canvas.width, canvas.height);
   725→            for (let p of particles) {
   726→                p.x += p.speedX;
   727→                p.y += p.speedY;
   728→                p.opacity += p.opacityDir * p.opacitySpeed;
   729→                if (p.opacity >= 0.8 || p.opacity <= 0.1) p.opacityDir *= -1;
   730→                p.rotation += p.rotationSpeed;
   731→
   732→                if (p.x < -10) p.x = canvas.width + 10;
   733→                if (p.x > canvas.width + 10) p.x = -10;
   734→                if (p.y < -10) p.y = canvas.height + 10;
   735→                if (p.y > canvas.height + 10) p.y = -10;
   736→
   737→                ctx.save();
   738→                ctx.translate(p.x, p.y);
   739→                ctx.rotate(p.rotation);
   740→                ctx.globalAlpha = p.opacity;
   741→
   742→                if (p.isStar) {
   743→                    ctx.fillStyle = '#FFD700';
   744→                    drawStar(0, 0, 4, p.size, p.size * 0.4);
   745→                } else {
   746→                    ctx.beginPath();
   747→                    ctx.arc(0, 0, Math.max(0.5, p.size), 0, Math.PI * 2);
   748→                    ctx.fillStyle = '#D4AF37';
   749→                    ctx.fill();
   750→                }
   751→                ctx.restore();
   752→            }
   753→            requestAnimationFrame(animateParticles);
   754→        }
   755→        animateParticles();

   765→            entries.forEach(function(entry) {
   766→                if (entry.isIntersecting) {

   777→        /* 3. COUNTDOWN TIMER (localStorage persisted)   */
   778→        /* ============================================ */
   779→        const COUNTDOWN_KEY = 'sbi_countdown_end';
   780→        const COUNTDOWN_MS = 72 * 60 * 60 * 1000;
   781→        let endTime = localStorage.getItem(COUNTDOWN_KEY);
   782→
   783→        if (!endTime || parseInt(endTime) < Date.now()) {
   784→            endTime = Date.now() + COUNTDOWN_MS;
   785→            localStorage.setItem(COUNTDOWN_KEY, endTime.toString());
   786→        } else {
   787→            endTime = parseInt(endTime);
   788→        }
   789→
   790→        const cdH = document.getElementById('cd-hours');
   791→        const cdM = document.getElementById('cd-minutes');
   792→        const cdS = document.getElementById('cd-seconds');
   793→
   794→        function updateCountdown() {
   795→            const now = Date.now();
   796→            let diff = endTime - now;
   797→            if (diff <= 0) { diff = 0; }
   798→            const h = Math.floor(diff / 3600000);
   799→            const m = Math.floor((diff % 3600000) / 60000);
   800→            const s = Math.floor((diff % 60000) / 1000);
   801→            cdH.textContent = String(h).padStart(2, '0');
   802→            cdM.textContent = String(m).padStart(2, '0');
   803→            cdS.textContent = String(s).padStart(2, '0');
   804→        }
   805→        updateCountdown();
   806→        setInterval(updateCountdown, 1000);
